# ls_cholesky_solver checkin test suite
# Solves Ax = y for symmetric positive definite A.
# Parameters: AIE_VARIANT DATA_TYPE DIM_SIZE GRID_DIM CASC_LEN NUM_FRAMES NITER

# Dimension constraints:
#   float  on AIE1/2  : DIM_SIZE must be a multiple of 8  (SIMD vector = 8 floats)
#   cfloat on AIE1/2  : DIM_SIZE must be a multiple of 4  (SIMD vector = 4 cfloats)
#   float  on AIE22   : DIM_SIZE must be a multiple of 16 (SIMD vector = 16 floats)
#   cfloat on AIE22   : DIM_SIZE must be a multiple of 8  (SIMD vector = 8 cfloats)

# --- AIE1 (vck190) ---

make all AIE_VARIANT=1  DATA_TYPE=cfloat DIM_SIZE=[32,64]    GRID_DIM=1 CASC_LEN=[2,4] NUM_FRAMES=1 NITER=4 DIFF_TOLERANCE=0.02 CC_TOLERANCE=0.02

make all AIE_VARIANT=1 DATA_TYPE=float  DIM_SIZE=64          GRID_DIM=1 CASC_LEN=4 NUM_FRAMES=1 NITER=4 DIFF_TOLERANCE=0.02 CC_TOLERANCE=0.02
